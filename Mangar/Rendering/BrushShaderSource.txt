#include <metal_stdlib>
using namespace metal;

struct Dab {
    float2 center;
    float radius;
    float opacity;
};

struct StampUniforms {
    float4 color;
    float2 tileOrigin;
    uint dabCount;
    uint mode;
};

kernel void stampRound(texture2d<float, access::read> source [[texture(0)]],
                       texture2d<float, access::write> output [[texture(1)]],
                       texture2d<float, access::read> selection [[texture(2)]],
                       constant Dab *dabs [[buffer(0)]],
                       constant StampUniforms &u [[buffer(1)]],
                       uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= source.get_width() || gid.y >= source.get_height()) return;
    float2 p = u.tileOrigin + float2(gid) + 0.5;
    float accumulated = 0.0;
    float mask = selection.read(gid).r;
    for (uint i = 0; i < u.dabCount; ++i) {
        float coverage = clamp(dabs[i].radius + 0.5 - distance(p, dabs[i].center), 0.0, 1.0);
        float a = coverage * dabs[i].opacity * mask;
        accumulated = 1.0 - (1.0 - accumulated) * (1.0 - a);
    }
    float4 old = source.read(gid);
    if (accumulated <= 0.0) {
        output.write(old, gid);
        return;
    }
    if (u.mode == 1) {
        output.write(old * (1.0 - accumulated), gid);
        return;
    }
    float4 ink = float4(u.color.rgb * accumulated, accumulated);
    output.write(ink + old * (1.0 - accumulated), gid);
}

kernel void stampRoundNonAccumulating(texture2d<float, access::read> base [[texture(0)]],
                                      texture2d<float, access::read> priorCoverage [[texture(1)]],
                                      texture2d<float, access::write> output [[texture(2)]],
                                      texture2d<float, access::write> nextCoverage [[texture(3)]],
                                      texture2d<float, access::read> selection [[texture(4)]],
                                      constant Dab *dabs [[buffer(0)]],
                                      constant StampUniforms &u [[buffer(1)]],
                                      uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= base.get_width() || gid.y >= base.get_height()) return;
    float2 p = u.tileOrigin + float2(gid) + 0.5;
    float coverage = priorCoverage.read(gid).r;
    for (uint i = 0; i < u.dabCount; ++i) {
        float edge = clamp(dabs[i].radius + 0.5 - distance(p, dabs[i].center), 0.0, 1.0);
        coverage = max(coverage, edge * dabs[i].opacity);
    }
    coverage = round(clamp(coverage, 0.0, 1.0) * 255.0) / 255.0;
    nextCoverage.write(float4(coverage, 0.0, 0.0, 1.0), gid);
    coverage *= selection.read(gid).r;
    float4 old = base.read(gid);
    if (u.mode == 1) {
        output.write(old * (1.0 - coverage), gid);
    } else {
        float4 ink = float4(u.color.rgb * coverage, coverage);
        output.write(ink + old * (1.0 - coverage), gid);
    }
}

struct SelectionOpUniforms {
    float4 rect;
    float2 tileOrigin;
    uint operation;
    uint padding;
};

kernel void editSelection(texture2d<float, access::read> previous [[texture(0)]],
                          texture2d<float, access::write> output [[texture(1)]],
                          constant SelectionOpUniforms &u [[buffer(0)]],
                          texture2d<float, access::read> incoming [[texture(2)]],
                          uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= output.get_width() || gid.y >= output.get_height()) return;
    float old = previous.read(gid).r;
    float2 pixel = u.tileOrigin + float2(gid);
    float overlapX = max(0.0, min(pixel.x + 1.0, u.rect.z) - max(pixel.x, u.rect.x));
    float overlapY = max(0.0, min(pixel.y + 1.0, u.rect.w) - max(pixel.y, u.rect.y));
    float value = u.padding == 1 ? incoming.read(gid).r : overlapX * overlapY;
    if (u.operation == 1) value = max(old, value);
    if (u.operation == 2) value = old * (1.0 - value);
    if (u.operation == 3) value = old * value;
    if (u.operation == 4) value = 1.0 - old;
    output.write(float4(value, 0.0, 0.0, 1.0), gid);
}

kernel void selectionTileBounds(texture2d<float, access::read> mask [[texture(0)]],
                                device atomic_uint *bounds [[buffer(0)]],
                                uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= mask.get_width() || gid.y >= mask.get_height()) return;
    if (mask.read(gid).r <= 0.0) return;
    atomic_fetch_min_explicit(&bounds[0], gid.x, memory_order_relaxed);
    atomic_fetch_min_explicit(&bounds[1], gid.y, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[2], gid.x + 1, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[3], gid.y + 1, memory_order_relaxed);
}

kernel void paintTileBounds(texture2d<float, access::read> tile [[texture(0)]],
                            device atomic_uint *bounds [[buffer(0)]],
                            uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= tile.get_width() || gid.y >= tile.get_height()) return;
    if (tile.read(gid).a <= 0.0) return;
    atomic_fetch_min_explicit(&bounds[0], gid.x, memory_order_relaxed);
    atomic_fetch_min_explicit(&bounds[1], gid.y, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[2], gid.x + 1, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[3], gid.y + 1, memory_order_relaxed);
}

struct QuadUniforms {
    float2 origin;
    float2 size;
    float2 viewport;
    float zoom;
    float rotation;
    float2 pan;
};

struct QuadOutput {
    float4 position [[position]];
    float2 uv;
    float2 canvasPosition;
};

vertex QuadOutput tileVertex(uint vertexID [[vertex_id]],
                             constant QuadUniforms &u [[buffer(0)]]) {
    const float2 corners[6] = {
        float2(0, 0), float2(1, 0), float2(0, 1),
        float2(0, 1), float2(1, 0), float2(1, 1)
    };
    float2 uv = corners[vertexID];
    float2 canvasPosition = u.origin + uv * u.size;
    float2 scaled = canvasPosition * u.zoom;
    float c = cos(u.rotation);
    float s = sin(u.rotation);
    float2 screen = float2(scaled.x * c - scaled.y * s,
                           scaled.x * s + scaled.y * c) + u.pan;
    float2 clip = float2(screen.x / u.viewport.x * 2.0 - 1.0,
                         1.0 - screen.y / u.viewport.y * 2.0);
    return { float4(clip, 0, 1), uv, canvasPosition };
}

fragment float4 tileFragment(QuadOutput in [[stage_in]],
                             texture2d<float> tile [[texture(0)]]) {
    constexpr sampler s(coord::normalized, address::clamp_to_edge, filter::nearest);
    return tile.sample(s, in.uv);
}

fragment float4 shapeFragment(QuadOutput in [[stage_in]], constant float4 &color [[buffer(0)]]) {
    return color;
}

fragment float4 checkerFragment(QuadOutput in [[stage_in]]) {
    int2 cell = int2(floor(in.canvasPosition / 16.0));
    float value = ((cell.x + cell.y) & 1) == 0 ? 0.85 : 0.96;
    return float4(value, value, value, 1.0);
}

struct GuideUniforms {
    float4 rect;
    float4 zoomAndPadding;
    float4 color;
};

fragment float4 pageGuideFragment(QuadOutput in [[stage_in]],
                                  constant GuideUniforms &u [[buffer(0)]]) {
    float2 p = in.canvasPosition;
    float padding = max(1.0, 1.0 / max(u.zoomAndPadding.x, 0.001));
    if (p.x < u.rect.x - padding || p.x > u.rect.z + padding ||
        p.y < u.rect.y - padding || p.y > u.rect.w + padding) discard_fragment();
    float distancePixels = min(min(abs(p.x - u.rect.x), abs(p.x - u.rect.z)),
                               min(abs(p.y - u.rect.y), abs(p.y - u.rect.w))) * u.zoomAndPadding.x;
    if (distancePixels > 0.8) discard_fragment();
    return u.color;
}

struct SelectionOutlineUniforms {
    float2 tileOrigin;
    float2 canvasSize;
    float zoom;
    float phase;
};

float selectionValue(int2 pixel, int2 local, int direction,
                     constant SelectionOutlineUniforms &u,
                     texture2d<float> center, texture2d<float> left,
                     texture2d<float> right, texture2d<float> top,
                     texture2d<float> bottom) {
    if (pixel.x < 0 || pixel.y < 0 || pixel.x >= int(u.canvasSize.x) ||
        pixel.y >= int(u.canvasSize.y)) return 0.0;
    if (direction == 1) return left.read(uint2(255, local.y)).r;
    if (direction == 2) return right.read(uint2(0, local.y)).r;
    if (direction == 3) return top.read(uint2(local.x, 255)).r;
    if (direction == 4) return bottom.read(uint2(local.x, 0)).r;
    return center.read(uint2(local)).r;
}

float4 marchingAntColor(float2 screen, float phase) {
    float stripe = floor((screen.x + screen.y) / 5.0 + phase);
    float shade = fmod(stripe, 2.0) == 0.0 ? 1.0 : 0.0;
    return float4(shade, shade, shade, 1.0);
}

fragment float4 selectionOutlineFragment(QuadOutput in [[stage_in]],
    texture2d<float> center [[texture(0)]], texture2d<float> left [[texture(1)]],
    texture2d<float> right [[texture(2)]], texture2d<float> top [[texture(3)]],
    texture2d<float> bottom [[texture(4)]],
    constant SelectionOutlineUniforms &u [[buffer(0)]]) {
    int2 pixel = int2(floor(in.canvasPosition));
    int2 local = pixel - int2(u.tileOrigin);
    if (local.x < 0 || local.y < 0 || local.x >= 256 || local.y >= 256) discard_fragment();
    bool inside = selectionValue(pixel, local, 0, u, center, left, right, top, bottom) > 0.5;
    float2 fraction = in.canvasPosition - float2(pixel);
    float distancePixels = 1e10;
    int leftDirection = local.x == 0 ? 1 : 0;
    int rightDirection = local.x == 255 ? 2 : 0;
    int topDirection = local.y == 0 ? 3 : 0;
    int bottomDirection = local.y == 255 ? 4 : 0;
    if (inside != (selectionValue(pixel + int2(-1, 0), local + int2(-1, 0),
                                  leftDirection, u, center, left, right, top, bottom) > 0.5))
        distancePixels = min(distancePixels, fraction.x * u.zoom);
    if (inside != (selectionValue(pixel + int2(1, 0), local + int2(1, 0),
                                  rightDirection, u, center, left, right, top, bottom) > 0.5))
        distancePixels = min(distancePixels, (1.0 - fraction.x) * u.zoom);
    if (inside != (selectionValue(pixel + int2(0, -1), local + int2(0, -1),
                                  topDirection, u, center, left, right, top, bottom) > 0.5))
        distancePixels = min(distancePixels, fraction.y * u.zoom);
    if (inside != (selectionValue(pixel + int2(0, 1), local + int2(0, 1),
                                  bottomDirection, u, center, left, right, top, bottom) > 0.5))
        distancePixels = min(distancePixels, (1.0 - fraction.y) * u.zoom);
    if (distancePixels > 0.75) discard_fragment();
    return marchingAntColor(in.position.xy, u.phase);
}

struct RectOutlineUniforms {
    float4 rect;
    float zoom;
    float phase;
};

fragment float4 selectionRectOutlineFragment(QuadOutput in [[stage_in]],
                                             constant RectOutlineUniforms &u [[buffer(0)]]) {
    float2 p = in.canvasPosition;
    if (p.x < u.rect.x - 1.0 || p.x > u.rect.z + 1.0 ||
        p.y < u.rect.y - 1.0 || p.y > u.rect.w + 1.0) discard_fragment();
    float distancePixels = min(min(abs(p.x - u.rect.x), abs(p.x - u.rect.z)),
                               min(abs(p.y - u.rect.y), abs(p.y - u.rect.w))) * u.zoom;
    if (distancePixels > 0.75) discard_fragment();
    return marchingAntColor(in.position.xy, u.phase);
}

// Programmable blending reads premultiplied destination color from tile memory.
struct LayerCompositeUniforms {
    float opacity;
    uint mode;
    uint solid;
    uint padding;
    float4 tone; // period in canvas pixels, angle in radians, density, enabled
    float4 canvasMap; // cos(rotation)/zoom, sin(rotation)/zoom, view pan x/y
    float4 solidColor; // Premultiplied graphic fill; mask visualization uses white.
};
struct LayerCompositeOutput {
    float4 color [[color(0), raster_order_group(0)]];
};
float3 layerBlendColor(float3 b, float3 s, uint mode) {
    switch (mode) {
        case 1: return b * s;
        case 2: return b + s - b * s;
        case 3: return select(2.0 * b * s, 1.0 - 2.0 * (1.0 - b) * (1.0 - s), b > 0.5);
        case 4: return min(b, s);
        case 5: return max(b, s);
        case 6: {
            float3 d = select(((16.0 * b - 12.0) * b + 4.0) * b, sqrt(b), b > 0.25);
            return select(b - (1.0 - 2.0 * s) * b * (1.0 - b),
                          b + (2.0 * s - 1.0) * (d - b), s > 0.5);
        }
        case 7: return select(2.0 * b * s, 1.0 - 2.0 * (1.0 - b) * (1.0 - s), s > 0.5);
        case 8: return abs(b - s);
        case 9: return min(float3(1.0), b + s);
        default: return s;
    }
}
float4 compositeLayerColor(float4 source, float4 destination, constant LayerCompositeUniforms &u) {
    source *= u.opacity;
    float3 b = destination.rgb / max(destination.a, 0.000001);
    float3 s = source.rgb / max(source.a, 0.000001);
    float3 blended = layerBlendColor(b, s, u.mode);
    float3 rgb = (1.0 - source.a) * destination.rgb +
                 (1.0 - destination.a) * source.rgb + source.a * destination.a * blended;
    return float4(rgb, source.a + destination.a * (1.0 - source.a));
}
float4 applyLayerTone(float4 source, float2 screen, constant LayerCompositeUniforms &u) {
    if (u.tone.w < 0.5) return source;
    float2 delta = screen - u.canvasMap.zw;
    float2 canvas = float2(delta.x * u.canvasMap.x + delta.y * u.canvasMap.y,
                          -delta.x * u.canvasMap.y + delta.y * u.canvasMap.x);
    float c = cos(u.tone.y), s = sin(u.tone.y);
    float2 grid = float2(canvas.x * c + canvas.y * s, -canvas.x * s + canvas.y * c) / u.tone.x;
    float distance = length(fract(grid) - 0.5);
    // Uniform scale gives an analytic one-screen-pixel footprint. Unlike quad
    // derivatives, this stays identical when panning by an odd pixel offset.
    float footprint = length(u.canvasMap.xy) / u.tone.x;
    float aa = max(footprint, 0.0001);
    float3 straight = source.rgb / max(source.a, 0.000001);
    float luminance = dot(straight, float3(0.2126, 0.7152, 0.0722));
    float coverage = saturate((1.0 - luminance) * source.a * u.tone.z);
    if (coverage <= 0.0) return float4(0.0);
    if (coverage >= 1.0) return float4(0.0, 0.0, 0.0, 1.0);
    // Circular dots grow into the cell corners at high concentrations.
    float radius = coverage <= 0.785398 ? sqrt(coverage / 3.141593) :
        mix(0.5, 0.707107, (coverage - 0.785398) / (1.0 - 0.785398));
    float mask = 1.0 - smoothstep(radius - aa * 0.5, radius + aa * 0.5, distance);
    // Below screen resolution, average coverage instead of producing moire.
    mask = mix(mask, coverage, smoothstep(0.6, 1.2, footprint));
    return float4(0.0, 0.0, 0.0, mask);
}

fragment LayerCompositeOutput layerCompositeFragment(
    QuadOutput in [[stage_in]], texture2d<float> tile [[texture(0)]],
    texture2d<float, access::read> layerMask [[texture(2)]],
    constant LayerCompositeUniforms &u [[buffer(0)]],
    float4 destination [[color(0), raster_order_group(0)]]) {
    constexpr sampler samplerNearest(coord::normalized, address::clamp_to_edge, filter::nearest);
    float4 source = u.solid != 0 ? u.solidColor : tile.sample(samplerNearest, in.uv);
    source = applyLayerTone(source, in.position.xy, u);
    if ((u.padding & 1u) != 0) source *= layerMask.read(uint2(in.position.xy)).r;
    if ((u.padding & 2u) != 0) source.rgb = float3(0.10, 0.45, 1.0) * source.a;
    return { compositeLayerColor(source, destination, u) };
}
// Compatibility path: source tiles never overlap within a layer, so one immutable
// backdrop copy per layer is sufficient on GPUs without raster order groups.
fragment float4 layerCompositeFallbackFragment(
    QuadOutput in [[stage_in]], texture2d<float> tile [[texture(0)]],
    texture2d<float> backdrop [[texture(1)]],
    texture2d<float, access::read> layerMask [[texture(2)]],
    constant LayerCompositeUniforms &u [[buffer(0)]]) {
    constexpr sampler samplerNearest(coord::normalized, address::clamp_to_edge, filter::nearest);
    float4 source = u.solid != 0 ? u.solidColor : tile.sample(samplerNearest, in.uv);
    source = applyLayerTone(source, in.position.xy, u);
    if ((u.padding & 1u) != 0) source *= layerMask.read(uint2(in.position.xy)).r;
    if ((u.padding & 2u) != 0) source.rgb = float3(0.10, 0.45, 1.0) * source.a;
    return compositeLayerColor(source, backdrop.read(uint2(in.position.xy)), u);
}

// Even-odd polygon coverage is produced by toggling one stencil bit per fan triangle.
vertex float4 polygonMaskVertex(uint id [[vertex_id]],
                                device const float2 *points [[buffer(0)]],
                                constant float4 &tile [[buffer(1)]]) {
    float2 local = (points[id] - tile.xy) / tile.zw;
    return float4(local.x * 2.0 - 1.0, 1.0 - local.y * 2.0, 0, 1);
}
fragment float4 polygonMaskFragment() { return float4(1); }

kernel void fillLasso(texture2d<float, access::read> base [[texture(0)]],
                      texture2d<float, access::read> shape [[texture(1)]],
                      texture2d<float, access::read> selection [[texture(2)]],
                      texture2d<float, access::write> output [[texture(3)]],
                      constant float4 &color [[buffer(0)]],
                      device atomic_uint *bounds [[buffer(1)]],
                      constant uint &erases [[buffer(2)]],
                      uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= output.get_width() || gid.y >= output.get_height()) return;
    float a = shape.read(gid).r * selection.read(gid).r * color.a;
    float4 old = base.read(gid);
    output.write((erases != 0 ? float4(0.0) : float4(color.rgb * a, a)) + old * (1.0 - a), gid);
    if (a <= 0.0) return;
    atomic_fetch_min_explicit(&bounds[0], gid.x, memory_order_relaxed);
    atomic_fetch_min_explicit(&bounds[1], gid.y, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[2], gid.x + 1, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[3], gid.y + 1, memory_order_relaxed);
}

struct LassoLineOutput { float4 position [[position]]; };
vertex LassoLineOutput lassoLineVertex(uint id [[vertex_id]],
                                      device const float2 *points [[buffer(0)]],
                                      constant QuadUniforms &u [[buffer(1)]]) {
    uint segment = id / 6;
    const float2 corners[6] = {float2(0,-1), float2(1,-1), float2(0,1),
                              float2(0,1), float2(1,-1), float2(1,1)};
    float c = cos(u.rotation), s = sin(u.rotation);
    float2 a = points[segment] * u.zoom, b = points[segment + 1] * u.zoom;
    a = float2(a.x*c-a.y*s, a.x*s+a.y*c) + u.pan;
    b = float2(b.x*c-b.y*s, b.x*s+b.y*c) + u.pan;
    float2 delta = b - a;
    float2 normal = float2(-delta.y, delta.x) / max(length(delta), 0.001);
    float2 p = mix(a, b, corners[id % 6].x) + normal * corners[id % 6].y;
    return {float4(p.x / u.viewport.x * 2.0 - 1.0, 1.0 - p.y / u.viewport.y * 2.0, 0, 1)};
}
fragment float4 lassoLineFragment(LassoLineOutput in [[stage_in]], constant float &phase [[buffer(0)]]) {
    float stripe = fmod(floor((in.position.x + in.position.y) / 5.0 + phase), 2.0);
    return float4(float3(stripe < 1.0 ? 0.0 : 1.0), 1);
}

kernel void stampLayerMask(texture2d<float, access::read> source [[texture(0)]],
                       texture2d<float, access::write> output [[texture(1)]],
                       texture2d<float, access::read> selection [[texture(2)]],
                       constant Dab *dabs [[buffer(0)]],
                       constant StampUniforms &u [[buffer(1)]],
                       uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= source.get_width() || gid.y >= source.get_height()) return;
    float2 p = u.tileOrigin + float2(gid) + 0.5;
    float accumulated = 0.0;
    float mask = selection.read(gid).r;
    for (uint i = 0; i < u.dabCount; ++i) {
        float coverage = clamp(dabs[i].radius + 0.5 - distance(p, dabs[i].center), 0.0, 1.0);
        float a = coverage * dabs[i].opacity * mask;
        accumulated = 1.0 - (1.0 - accumulated) * (1.0 - a);
    }
    float old = source.read(gid).r;
    if (accumulated <= 0.0) {
        output.write(float4(old, 0, 0, 1), gid);
        return;
    }
    float gray = u.mode == 1 ? 1.0 : u.color.r;
    output.write(float4(mix(old, gray, accumulated), 0, 0, 1), gid);
}

kernel void stampLayerMaskNonAccumulating(texture2d<float, access::read> base [[texture(0)]],
                                      texture2d<float, access::read> priorCoverage [[texture(1)]],
                                      texture2d<float, access::write> output [[texture(2)]],
                                      texture2d<float, access::write> nextCoverage [[texture(3)]],
                                      texture2d<float, access::read> selection [[texture(4)]],
                                      constant Dab *dabs [[buffer(0)]],
                                      constant StampUniforms &u [[buffer(1)]],
                                      uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= base.get_width() || gid.y >= base.get_height()) return;
    float2 p = u.tileOrigin + float2(gid) + 0.5;
    float coverage = priorCoverage.read(gid).r;
    for (uint i = 0; i < u.dabCount; ++i) {
        float edge = clamp(dabs[i].radius + 0.5 - distance(p, dabs[i].center), 0.0, 1.0);
        coverage = max(coverage, edge * dabs[i].opacity);
    }
    coverage = round(clamp(coverage, 0.0, 1.0) * 255.0) / 255.0;
    nextCoverage.write(float4(coverage, 0.0, 0.0, 1.0), gid);
    coverage *= selection.read(gid).r;
    float old = base.read(gid).r;
    float gray = u.mode == 1 ? 1.0 : u.color.r;
    output.write(float4(mix(old, gray, coverage), 0, 0, 1), gid);
}

kernel void fillLassoLayerMask(texture2d<float, access::read> base [[texture(0)]],
                      texture2d<float, access::read> shape [[texture(1)]],
                      texture2d<float, access::read> selection [[texture(2)]],
                      texture2d<float, access::write> output [[texture(3)]],
                      constant float4 &color [[buffer(0)]],
                      device atomic_uint *bounds [[buffer(1)]],
                      constant uint &erases [[buffer(2)]],
                      uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= output.get_width() || gid.y >= output.get_height()) return;
    float a = shape.read(gid).r * selection.read(gid).r * color.a;
    float old = base.read(gid).r;
    output.write(float4(mix(old, erases != 0 ? 1.0 : color.r, a), 0, 0, 1), gid);
    if (a <= 0.0) return;
    atomic_fetch_min_explicit(&bounds[0], gid.x, memory_order_relaxed);
    atomic_fetch_min_explicit(&bounds[1], gid.y, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[2], gid.x + 1, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[3], gid.y + 1, memory_order_relaxed);
}

}

// Clear respects fractional selection coverage and keeps published source tiles immutable.
kernel void clearLayer(texture2d<float, access::read> source [[texture(0)]],
                       texture2d<float, access::read> selection [[texture(1)]],
                       texture2d<float, access::write> output [[texture(2)]],
                       constant uint &isMask [[buffer(0)]],
                       device atomic_uint *bounds [[buffer(1)]],
                       uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= output.get_width() || gid.y >= output.get_height()) return;
    float coverage = selection.read(gid).r;
    float4 old = source.read(gid);
    float4 value = isMask != 0 ? float4(mix(old.r, 1.0, coverage), 0, 0, 1)
                              : old * (1.0 - coverage);
    output.write(value, gid);
    bool changed = isMask != 0 ? value.r != old.r : any(value != old);
    if (!changed) return;
    atomic_fetch_min_explicit(&bounds[0], gid.x, memory_order_relaxed);
    atomic_fetch_min_explicit(&bounds[1], gid.y, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[2], gid.x + 1, memory_order_relaxed);
    atomic_fetch_max_explicit(&bounds[3], gid.y + 1, memory_order_relaxed);
}
